Set-Location -LiteralPath $Repo

# El repositorio debe estar limpio.
$Cambios = git status --porcelain

if (-not [string]::IsNullOrWhiteSpace(($Cambios | Out-String))) {
    Write-Host 'API Reports contiene cambios locales:' -ForegroundColor Red
    git status --short
    throw 'No se continuará hasta que el repositorio esté limpio.'
}

git fetch origin --prune
if ($LASTEXITCODE -ne 0) {
    throw 'Falló git fetch.'
}

$OrigenRemoto = "origin/$RamaOrigen"
$DestinoRemoto = "origin/$RamaDestino"

git show-ref --verify --quiet "refs/remotes/$OrigenRemoto"
if ($LASTEXITCODE -ne 0) {
    throw "No existe $OrigenRemoto."
}

git show-ref --verify --quiet "refs/remotes/$DestinoRemoto"
if ($LASTEXITCODE -ne 0) {
    throw "No existe $DestinoRemoto. La rama destino debe crearse primero en GitHub."
}

# Crear localmente PRE-MAIN-REPORTS desde su correspondiente rama remota.
git show-ref --verify --quiet "refs/heads/$RamaDestino"

if ($LASTEXITCODE -eq 0) {
    git switch $RamaDestino
}
else {
    git switch --track -c $RamaDestino $DestinoRemoto
}

if ($LASTEXITCODE -ne 0) {
    throw "No fue posible posicionarse en $RamaDestino."
}

git pull --ff-only origin $RamaDestino
if ($LASTEXITCODE -ne 0) {
    throw "No fue posible actualizar $RamaDestino."
}

Write-Host ''
Write-Host "Se aplicarán $($Commits.Count) commits de API Reports." -ForegroundColor Cyan

foreach ($Commit in $Commits) {
    git cat-file -e "${Commit}^{commit}"

    if ($LASTEXITCODE -ne 0) {
        throw "No existe el commit $Commit."
    }

    git merge-base --is-ancestor $Commit $OrigenRemoto

    if ($LASTEXITCODE -ne 0) {
        throw "El commit $Commit no pertenece a $OrigenRemoto."
    }

    $Padres = @(
        (git rev-list --parents -n 1 $Commit) -split '\s+'
    )

    if ($Padres.Count -ne 2) {
        throw "El commit $Commit es un merge y no será aplicado."
    }

    git --no-pager show `
        --no-patch `
        --format='%h | %an | %s' `
        $Commit
}

$Confirmacion = Read-Host "`nEscribe APLICAR API REPORTS"

if ($Confirmacion -ne 'APLICAR API REPORTS') {
    throw 'Proceso cancelado.'
}

foreach ($Commit in $Commits) {
    Write-Host "Aplicando $Commit..." -ForegroundColor Cyan

    git cherry-pick --allow-empty $Commit

    if ($LASTEXITCODE -ne 0) {
        Write-Host ''
        Write-Host "Conflicto en $Commit. No se hizo push." -ForegroundColor Red
        git status --short
        throw "Se detuvo el cherry-pick en $Commit."
    }
}

$EstadoFinal = git status --porcelain

if (-not [string]::IsNullOrWhiteSpace(($EstadoFinal | Out-String))) {
    throw 'API Reports terminó con cambios pendientes; no se hará push.'
}

Write-Host ''
Write-Host 'Commits pendientes de publicación:' -ForegroundColor Cyan
git --no-pager log --oneline "$DestinoRemoto..$RamaDestino"

$ConfirmarPush = Read-Host 'Escribe PUBLICAR API REPORTS'

if ($ConfirmarPush -ne 'PUBLICAR API REPORTS') {
    throw 'Los commits quedaron locales, pero no se hizo push.'
}

git push -u origin "$RamaDestino`:$RamaDestino"

if ($LASTEXITCODE -ne 0) {
    throw 'Falló el push de API Reports.'
}

Write-Host ''
Write-Host 'API REPORTS publicado correctamente.' -ForegroundColor Green
