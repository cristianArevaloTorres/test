        }

        $RamaDespues = (git branch --show-current).Trim()

        if ($RamaAntes -ne $RamaDespues) {
            throw "La rama local cambió inesperadamente de $RamaAntes a $RamaDespues."
        }

        Write-Host ''
        Write-Host "Últimos commits descargados de origin/$($Repo.RamaOrigen):" -ForegroundColor Green

        git log `
            "origin/$($Repo.RamaOrigen)" `
            --first-parent `
            --no-merges `
            -15 `

            --pretty=format:'%h | %an | %ad | %s'
